function escapeHtml(s){return s.replace(/[&<>]/g,c=>({'&':'&amp;','<':'&lt;','>':'&gt;'}[c]));}
function highlight(code,lang){
  let lines=code.textContent.split('\n');
  code.innerHTML=lines.map(line=>{
    let e=escapeHtml(line);
    if(/^\s*#/.test(line)) return '<span class="com">'+e+'</span>';
    if(lang==='containerfile') e=e.replace(/^(FROM|ARG|RUN|COPY|LABEL|CMD|ENV|WORKDIR)\b/,'<span class="kw">$1</span>');
    else e=e.replace(/^(\s*)(sudo|export|printf|cat|echo|read|unset|rpm|uname|systemctl|sysctl|mkdir|grep)\b/,'$1<span class="cmd">$2</span>');
    e=e.replace(/(\$\{[^}]+\}|\$[A-Z_][A-Z0-9_]*)/g,'<span class="var">$1</span>');
    return e;
  }).join('\n');
}
async function copyText(text){
  // HTTPS/localhost 등 Clipboard API를 사용할 수 있는 환경
  if (navigator.clipboard && window.isSecureContext) {
    await navigator.clipboard.writeText(text);
    return;
  }

  // file:// 또는 Clipboard API가 제한된 환경용 fallback
  const ta = document.createElement('textarea');
  ta.value = text;
  ta.setAttribute('readonly', '');
  ta.style.position = 'fixed';
  ta.style.left = '-9999px';
  ta.style.top = '0';
  document.body.appendChild(ta);
  ta.focus();
  ta.select();

  const ok = document.execCommand('copy');
  document.body.removeChild(ta);

  if (!ok) throw new Error('copy command failed');
}
